// 21-ai-gguf-parse.txt —— 纯极语言解析 GGUF 量化模型头部（实测通过；故意不写 定义 位数=64;）
// 输出: GGUF v3 / tensors=320 / metadata_kv=35 / general.file_type=15(Q4_K_M) / 前 12 个张量 ggml_type
// 原因: 64 位产物下 "指针(偏移)" 单字节读会 0xC0000005，逐字节解析必须用 32 位产物
// 运行: powershell -File scripts\sec.ps1 -Src examples\21-ai-gguf-parse.txt -Run
引入 "lib\kernel32.lib";
导入 冲刷 别名 fflush 支持库 "MSVCRT.DLL",1;
整数 G数据=0;
整数 G长度=0;
整数 G位置=0;
文本 G串[512];
程序 初始启动
文本 路径[260];
复制文字(路径,"D:\workplace\Police-analysis\models\Qwen3.5-0.8B.Q4_K_M.gguf");
整数 文件=开文件(路径,0);
如果(文件<=0)
{
	写文字("打不开 GGUF 文件");写字符(13);写字符(10);
	退出(1);
}
整数 高位=0;
整数 全长=文件大小(文件,@高位);
整数 读长=16777216;
G数据=申请内存(读长+16);
G长度=读文件(文件,G数据,读长);
关文件(文件);
写格式("GGUF 文件总长 = %d 字节，读入头部 %d 字节\r\n",全长,G长度);
整数 魔=取四(0);
写格式("magic   = %08X   (GGUF = 46554747)\r\n",魔);
整数 版本=取四(4);
整数 张量数=取四(8);
整数 键值数=取四(16);
写格式("version = %d\r\n",版本);
写格式("tensors = %d    metadata_kv = %d\r\n",张量数,键值数);
冲刷(0);
G位置=24;
写文字("---- 1) 元数据键值 (键名 + 类型 + 值) ----");写字符(13);写字符(10);
整数 i=0;
循环于(i=0;i<键值数;i++)
{
	读串;
	写文字("  ");写文字(G串);
	整数 类型号=取四(G位置);
	G位置=G位置+4;
	写格式("  [类型%d] ",类型号);
	如果(类型号=8)
	{
		读串;
		写文字("= ");写文字(G串);
	}
	否则
	{
		如果(类型号=4){整数 v4=取四(G位置);写格式("= %d",v4);}
		如果(类型号=5){整数 v5=取四(G位置);写格式("= %d",v5);}
		如果(类型号=6){整数 v6=取四(G位置);写格式("= %08X (float)",v6);}
		如果(类型号=10){整数 v10=取四(G位置);写格式("= %d (u64低32位)",v10);}
		如果(类型号=9)
		{
			整数 元型=取四(G位置);
			整数 元素数=取四(G位置+4);
			写格式("array<类型%d> 共%d项",元型,元素数);
		}
		跳值(类型号);
	}
	写字符(13);写字符(10);
	冲刷(0);
}
写文字("---- 2) 张量清单（ggml_type 即量化类型号）----");写字符(13);写字符(10);
整数 j=0;
循环于(j=0;j<张量数;j++)
{
	如果(j>11){跳出;}
	读串;
	写文字("  ");写文字(G串);
	整数 维数=取四(G位置);
	G位置=G位置+4;
	打维度(维数);
	整数 张量型=取四(G位置);
	G位置=G位置+4;
	写格式("   ggml_type=%d",张量型);
	G位置=G位置+8;
	写字符(13);写字符(10);
	冲刷(0);
}
写文字("OK: 极语言 不依赖任何 DLL，直接解析了 GGUF(量化模型) 头部/元数据/张量表");写字符(13);写字符(10);
冲刷(0);
退出(0);
结束
程序 打维度(维数)
整数 k=0;
循环于(k=0;k<维数;k++)
{
	整数 维=取四(G位置);
	写格式("[%d]",维);
	G位置=G位置+8;
}
结束
程序 取四(偏)
整数 b0=G数据(偏);
整数 b1=G数据(偏+1);
整数 b2=G数据(偏+2);
整数 b3=G数据(偏+3);
返回(b0+b1*256+b2*65536+b3*16777216);
结束
程序 读串
整数 长=取四(G位置);
清零(G串,512);
如果(长>96)
{
	长=96;
}
如果(长>0)
{
	整数 源=G数据+G位置+8;
	内存复制(G串,源,长);
}
G串(长)=0;
G位置=G位置+8+取四(G位置);
结束
程序 跳串
整数 长=取四(G位置);
G位置=G位置+8+长;
结束
程序 跳元素(元型)
如果(元型=8)
{
	跳串;
	返回(0);
}
G位置=G位置+型长(元型);
结束
程序 跳值(类型号)
如果(类型号=8)
{
	跳串;
	返回(0);
}
如果(类型号=9)
{
	整数 元型=取四(G位置);
	整数 个数=取四(G位置+4);
	G位置=G位置+12;
	整数 m=0;
	循环于(m=0;m<个数;m++)
	{
		跳元素(元型);
	}
	返回(0);
}
G位置=G位置+型长(类型号);
结束
程序 型长(类型号)
如果(类型号=0){返回(1);}
如果(类型号=1){返回(1);}
如果(类型号=2){返回(2);}
如果(类型号=3){返回(2);}
如果(类型号=4){返回(4);}
如果(类型号=5){返回(4);}
如果(类型号=6){返回(4);}
如果(类型号=7){返回(1);}
如果(类型号=10){返回(8);}
如果(类型号=11){返回(8);}
如果(类型号=12){返回(8);}
返回(0);
结束
