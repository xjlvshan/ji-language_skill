// sys-4 进程与模块：进程快照/查找进程/下个进程/关闭对象 + 进程类字段 + 模块遍历
// 坑：进程类/模块类 必须写 (长度=296)/(长度=1316)；进程快照(2,0)=进程, (8,pid)=模块
文本 名[32];
进程类 进程 (长度=296);
模块类 模块 (长度=1316);
整数 快照;
整数 下个;
整数 计数;
整数 编号;
程序 初始启动
	写文字("=== 1. 进程遍历（进程快照(2,0) + 查找进程/下个进程）===");
	计数=0;
	快照=进程快照(2,0);
	写格式("进程快照(2,0) 返回=%d",快照);写字符(10);
	如果(快照<>0){
		下个=查找进程(快照,进程);
		循环{
			写格式("  PID=%-6d 线程=%d 父PID=%-6d 名称=",进程.编号,进程.线程,进程.父进程);
			写文字(进程.名称);
			计数++;
			如果(计数>=12){跳出}
			下个=下个进程(快照,进程);
		}(下个=0);
		写格式("共列出 %d 个进程（已截断）",计数);写字符(10);
		关闭对象(快照);
	}
	写字符(10);
	写文字("=== 2. 本进程信息 ===");
	写格式("进程标识=%d 当前进程句柄=%d 开机毫秒=%d",进程标识,当前进程,开机毫秒);写字符(10);
	模块名称(0,名,32);
	写文字("名(截断)=");写文字(名);
	写字符(10);
	写字符(10);
	写文字("=== 3. 模块遍历：进程快照(8,本进程PID) + 查找模块/下个模块 ===");
	快照=进程快照(8,进程标识);
	写格式("进程快照(8,pid) 返回=%d",快照);写字符(10);
	如果(快照<>0){
		计数=0;
		下个=查找模块(快照,模块);
		循环{
			写格式("  基址=0x%X 大小=%d 模块=",模块.基址,模块.大小);
			写文字(模块.名称);
			计数++;
			如果(计数>=8){跳出}
			下个=下个模块(快照,模块);
		}(下个=0);
		写格式("共列出 %d 个模块（已截断）",计数);写字符(10);
		关闭对象(快照);
	}
	写字符(10);
	写文字("=== 4. 打开进程/关闭对象 ===");
	编号=打开进程($1F0FFF,0,进程标识);
	写格式("打开进程(全部权限,本PID) 返回=%d",编号);写字符(10);
	写格式("关闭对象 返回=%d",关闭对象(编号));写字符(10);
	写文字("--- sys-4 演示结束 ---");
	退出(0);
结束
