// sys-2 信息框 / 延时框：按钮与图标常量、返回值（控制台程序实测）
// 坑：控制台程序里 信息框 会阻塞且控制台在后台，测试必须用 延时框(有超时自动关闭)
导入 信息框 别名 MessageBoxA 支持库 "USER32.DLL",4;
导入 延时框 别名 MessageBoxTimeoutA 支持库 "USER32.DLL",6;
程序 初始启动
	写文字("=== 1. 按钮常量（MB_* 低4位） ===");
	写格式("确定取消=$%X 中止重试=$%X 是否取消=$%X 弹出是否=$%X 重试取消=$%X",$1,$2,$3,$4,$5);写字符(10);
	写文字("=== 2. 图标常量 ===");
	写格式("错误图标=$%X 问号图标=$%X 惊叹图标=$%X 信息图标=$%X 用户图标=$%X",$10,$20,$30,$40,$80);写字符(10);
	写文字("=== 3. 控件常量 ===");
	写格式("弹出系统=$%X 弹出任务=$%X 帮助按钮=$%X 弹出置前=$%X 弹出置顶=$%X 文字居右=$%X 图标居右=$%X",$1000,$2000,$4000,$10000,$40000,$80000,$100000);写字符(10);
	写文字("=== 4. 返回值常量 ===");
	写格式("选了确定=%d 选了取消=%d 选了中止=%d 选了重试=%d 选了忽略=%d",1,2,3,4,5);写字符(10);
	写格式("选了是=%d 选了否=%d 选了关闭=%d 选了帮助=%d",6,7,8,9);写字符(10);
	写文字("=== 5. 组合样式（图标+按钮用 + 相加） ===");
	写格式("问号+确定取消=$%X",$20+$1);写字符(10);
	写格式("信息+弹出是否=$%X",$40+$4);写字符(10);
	写字符(10);
	写文字("=== 6. 延时框：3秒后自动关闭，不阻塞批处理 ===");
	整数 编号=延时框(0,"3秒后自动关闭","sys-2 延时框(测试)",[$20+$1],0,3000);
	写格式("延时框 返回=%d (32000=超时自动关闭)",编号);写字符(10);
	写字符(10);
	写文字("=== 7. 信息框：会阻塞直到用户点击（此处不调用）===");
	写文字("实测：调用 信息框(0,内容,标题,标志) 后程序停住，");
	写文字("      且控制台窗口在后台/被重定向时无人能点击 -> 只能超时杀进程");
	写文字("      结论：控制台程序想弹提示，用 延时框，或用 命令行 启动另一个进程");
	写文字("--- sys-2 演示结束 ---");
	退出(0);
