程序类型=2
模块文件=CMD.inc
引入 "lib\ws2_32.lib";
// sys-6 网络：网络启动/分配/转换地址/连网/发送/接收/关闭/网络终止
// 只探测本机 127.0.0.1，不连外网
文本 信息[400];
文本 缓冲[128];
整数 网号;
整数 结果;
整数 已收;
地址类 地址 (协议=2);
程序 初始启动
	写文字("=== 1. 网络启动(WSAStartup) ===");
	结果=网络启动($202,信息);
	写格式("网络启动(版本$202) 返回=%d (0=成功)",结果);写字符(10);
	写字符(10);
	写文字("=== 2. 分配(socket) ===");
	网号=分配(2,1,6);
	写格式("分配(网络型2=AF_INET,数据流1=SOCK_STREAM,流协议6=IPPROTO_TCP) 返回=%d",网号);写字符(10);
	如果(网号>=0){
		写字符(10);
		写文字("=== 3. 转换地址(inet_addr) ===");
		地址.地址=转换地址("127.0.0.1");
		写格式("转换地址(127.0.0.1)=%d",地址.地址);写字符(10);
		地址.端口=反转(59999);
		写格式("反转(59999)=%d",地址.端口);写字符(10);
		写字符(10);
		写文字("=== 4. 连网(connect) 本机未监听端口 59999 ===");
		结果=连网(网号,地址,16);
		写格式("连网 返回=%d (-1=失败，符合预期)",结果);写字符(10);
		写字符(10);
		写文字("=== 5. 发送/接收（连接失败时不调用）===");
		如果(结果=0){
			复制文字(缓冲,"PING");
			写格式("发送 返回=%d",发送(网号,缓冲,4,0));写字符(10);
			已收=接收(网号,缓冲,128,0);
			写格式("接收 返回=%d",已收);写字符(10);
		}否则{
			写文字("  连接失败，跳过 发送/接收");
		}
		写字符(10);
		写文字("=== 6. 关闭(closesocket) ===");
		写格式("关闭 返回=%d",关闭(网号));写字符(10);
	}
	写字符(10);
	写文字("=== 7. 成功连接对照：127.0.0.1:135 (Windows RPC 本机监听) ===");
	网号=分配(2,1,6);
	地址.地址=转换地址("127.0.0.1");
	地址.端口=反转(135);
	结果=连网(网号,地址,16);
	写格式("连网(127.0.0.1:135) 返回=%d (0=成功)",结果);写字符(10);
	如果(结果=0){
		复制文字(缓冲,"hello-tcp");
		写格式("  发送(10字节) 返回=%d",发送(网号,缓冲,10,0));写字符(10);
	}
	关闭(网号);
	写字符(10);
	写文字("=== 8. 网络终止(WSACleanup) ===");
	写格式("网络终止 返回=%d",网络终止);写字符(10);
	写文字("--- sys-6 演示结束 ---");
	退出(0);
结束
