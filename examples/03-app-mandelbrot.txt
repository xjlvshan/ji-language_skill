模块文件=CMD.inc
程序 初始启动
写文字("== 极语言 ASCII 曼德博集合 ==");
整数 色表=" .:-=+*#%@";
文本 行[96];
整数 列,行号,迭代,色号;
浮点 fx,fy,zx,zy,t;
循环于(行号=0;行号<22;行号++)
{
fy=行号;
fy=fy*0.11-1.2;
循环于(列=0;列<78;列++)
{
fx=列;
fx=fx*0.042-2.15;
zx=0.0;
zy=0.0;
迭代=0;
循环在(迭代<32)
{
t=zx*zx-zy*zy+fx;
zy=2.0*zx*zy+fy;
zx=t;
如果(zx*zx+zy*zy>4.0){跳出;}
迭代++;
}
色号=迭代 余 10;
行(列)=色表(色号);
}
行(78)=0;
写文字(行);
}
写文字("done.");
