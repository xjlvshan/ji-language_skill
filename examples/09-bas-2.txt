程序类型=2
模块文件=CMD.inc

/* 全局变量区：所有 程序/函数 之外
   注意：文本变量在全局声明时就带初始值是合法的（实测正常）；
   但在程序运行中写 文本变量="字面量"; 会死循环卡死，详见 bas-7。 */
文本 全局文本[32]="全局";
整数 全局数=5;
布尔 全局真假=真;
文本 缓冲[64];

程序 初始启动
	写文字("== bas-2 变量类型全表与赋值转换 ==");
	布尔 真假=真;
	字符 字=-3;
	字节 节=200;
	短数 短=1234;
	正短 正=60000;
	整数 数=-123456;
	变数 变=7;
	大数 大=2000000;
	大=大*5000;
	小数 单=1.5;
	浮点 浮=2.25;
	双精 双=3.125;
	写格式("布尔=%d 字符=%d 字节=%d 短数=%d 正短=%d\r\n",真假,字,节,短,正);
	写格式("整数=%d 变数=%d\r\n",数,变);
	缓冲(0)=0;
	缓冲=缓冲+"大数(64位) 2000000*5000 = "+大;
	写文字(缓冲);
	写文字("-- 小数/浮点/双精 的文本化（拼接最可靠） --");
	缓冲(0)=0;
	缓冲=缓冲+"单="+单+" 浮="+浮+" 双="+双;
	写文字(缓冲);
	写格式("双精变量能用 %%f 直接打印: %f\r\n",双);
	写格式("全局变量 全局数=%d 全局真假=%d\r\n",全局数,全局真假);
	写文字("-- 等号 = 按左边类型转换 --");
	整数 取整=单;
	写格式("小数 1.5 转整数 = %d（四舍六入五成双）\r\n",取整);
	整数 数二=7,数三=2;
	小数 商=数二/数三;
	缓冲(0)=0;
	缓冲=缓冲+"整数相除赋给小数 = "+商;
	写文字(缓冲);
	双精 双商=数二/数三;
	写格式("整数相除赋给双精 = %f\r\n",双商);
	数二=:数二-数三;
	写格式("冒号触发浮点运算 数二=:数二-数三 -> %d\r\n",数二);
	写文字("-- 数字自动转文本并拼接 --");
	缓冲(0)=0;
	缓冲=缓冲+"结果:"+数二;
	写文字(缓冲);
	写文字("程序结束");
结束
