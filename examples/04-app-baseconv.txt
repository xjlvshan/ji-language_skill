程序类型=2
模块文件=CMD.inc
程序 初始启动
整数 行=启动参数;
整数 i=0;
整数 末=0;
循环{
如果(行(i)=0){跳出;}
如果(行(i)=32){末=i+1;}
i++;
}
整数 值=0;
整数 j=末;
循环在(行(j)>=48){
如果(行(j)<=57){值=值*10+(行(j)-48);j++;}
否则{跳出;}
}
写格式("输入=%d\r\n",值);
整数 n=值;
整数 位=0;
文本 十六[32];
十六(0)=0;
循环{
整数 d=n 余 16;
n=n/16;
如果(d<10){十六(位)=48+d;}
否则{十六(位)=55+d;}
位++;
如果(n=0){跳出;}
}
写文字("十六进制=");
整数 k=位-1;
循环在(k>=0){写字符(十六(k));k--;}
写字符(13);写字符(10);
n=值;
位=0;
文本 二进[40];
二进(0)=0;
循环{
整数 b=n 余 2;
n=n/2;
二进(位)=48+b;
位++;
如果(n=0){跳出;}
}
写文字("二进制=");
k=位-1;
循环在(k>=0){写字符(二进(k));k--;}
写字符(13);写字符(10);
结束
