// 30-app-net-json —— 极语言实战：JSON 构造 + GBK<->UTF-8 + Winsock
// 本文件只用「实测可用」的写法：
//   * 所有局部声明写在过程最前面
//   * 不在 如果(...) 条件里调用自定义过程
//   * 不在过程里对「参数」做指针加减（改为全局缓冲 + 偏移）
//   * 死循环写 循环 / 循环{...} ，不要写 循环(1)
// 实测输出：JSON 里的引号被转义成 反斜杠+引号、反斜杠被转义成两个反斜杠；
//           GBK 67 字节 -> UTF-8 70 字节；长度前缀 00000070；连接到 9 端口返回 -1、135 端口返回 0。
// ============================================================

文本 报文[4096];
整数 报文位=0;
文本 请求UTF8[8192];
文本 宽缓存[16384];
文本 提示[512];
文本 方向字[16];
整数 转换长度=0;

程序 初始启动
	// ---- 1. 构造 JSON（只构造，不解析）----
	报重置;
	报文字("{");
	报引号;报文字("cmd");报引号;报文字(":");报引号;报文字("translate");报引号;报文字(",");
	报引号;报文字("text");报引号;报文字(":");报引号;
	报转义("He said ''hi'' \ 反斜杠");   // 注意：字符串字面量里用 '' 表示双引号
	报引号;报文字(",");
	报引号;报文字("dir");报引号;报文字(":");报引号;报文字("en_zh");报引号;
	报文字("}");
	写文字("JSON = ");
	写文字(报文);
	写字符(13);写字符(10);

	// ---- 2. GBK -> UTF-8 ----
	本地转UTF8(报文,报文位,请求UTF8,8192);
	写格式("GBK 字节=%d  UTF-8 字节=%d\r\n",报文位,转换长度);

	// ---- 3. 8 位十进制长度前缀（不要手写取余循环，见 SKILL 陷阱表）----
	文本 头缓冲[64];
	格式化(头缓冲,"%08d",转换长度);
	写格式("长度前缀=[%s]\r\n",头缓冲);

	// ---- 4. Winsock：连接探测 ----
	整数 结果=连接探测("127.0.0.1",9);
	写格式("连 127.0.0.1:9 = %d （-1 表示端口未开放，正常）\r\n",结果);
	结果=连接探测("127.0.0.1",135);
	写格式("连 127.0.0.1:135 = %d\r\n",结果);
	退出(0);
结束

// ---------- 报文构造 ----------
程序 报重置
	报文位=0;
	报文(0)=0;
结束

程序 报字节(码)
	报文(报文位)=码;
	报文位=报文位+1;
	报文(报文位)=0;
结束

程序 报文字(内容)
	整数 n;
	n=文字长度(内容);
	内存复制(报文+报文位,内容,n);
	报文位=报文位+n;
	报文(报文位)=0;
结束

程序 报引号
	报字节(34);
结束

// JSON 字符串转义：\ " 换行 回车
程序 报转义(内容)
	整数 i,c,转义,原样;
	i=0;
	循环
	{
		c=内容(i);
		如果(c=0){跳出}
		转义=0;
		原样=1;
		如果(c=34){转义=1}
		如果(c=92){转义=1}
		如果(c=10){转义=1}
		如果(c=13){转义=1}
		如果(c=10){原样=0}
		如果(c=13){原样=0}
		如果(转义=1){报字节(92)}
		如果(c=10){报字节(110)}
		如果(c=13){报字节(114)}
		如果(原样=1){报字节(c)}
		i++;
	}
结束

// ---------- 编码 ----------
程序 本地转UTF8(源,源长,目标,目标大小)
	整数 宽数,n;
	宽数=到宽字符(936,0,源,源长,0,0);
	如果(宽数<=0){目标(0)=0;转换长度=0;返回(-1)}
	到宽字符(936,0,源,源长,宽缓存,宽数);
	n=到多字符(65001,0,宽缓存,宽数,目标,目标大小,0,0);
	转换长度=n;
	返回(n)
结束

// ---------- Winsock 连接探测 ----------
程序 连接探测(网址,端口号)
	文本 数据[400];
	文本 地址[16];
	整数 启,网,连;
	启=网络启动($202,数据);
	网=分配(2,1,6);
	如果(网<0){返回(-1)}
	地址(0)%=2;
	地址(2)%=反转(端口号);
	地址(4)&=转换地址(网址);
	连=连网(网,地址,16);
	关闭(网);
	网络终止;
	返回(连)
结束
