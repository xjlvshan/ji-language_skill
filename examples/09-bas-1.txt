程序类型=2
模块文件=CMD.inc

// ===== 全局变量：写在第一个 程序/函数 之前 =====
整数 全局数=100;
文本 全局文本[32]="全局文本";

/* ===== 入口规则（本版最关键的一条） =====
   源码里第一个出现的 程序/函数 就是程序入口，
   它叫什么名字其实无所谓；辅助例程必须写在它后面。 */

程序 初始启动
	局部 整数 局部数=7;
	写文字("== bas-1 程序结构与入口 ==");
	写格式("全局数=%d 局部数=%d\r\n", 全局数, 局部数);
	写文字(全局文本);
	整数 和=加法(3,4);
	写格式("程序 加法(3,4) = %d\r\n", 和);
	和=乘法(6,7);
	写格式("函数 乘法(6,7) 大括号体 = %d\r\n", 和);
	显示(和);
	整数 a=1; 整数 b=2; 整数 c=3
	写格式("分号和换行都是语句分隔符 a=%d b=%d c=%d\r\n", a, b, c);
	全局数 加一;
	写格式("全局数 加一 后 = %d\r\n", 全局数);
	整数 缺省=加法();
	写格式("加法() 省略实参 = %d （形参默认值不生效，是垃圾值）\r\n", 缺省);
	写文字("程序结束");
结束

程序 加法(数一=10,数二=5)
	返回(数一 + 数二)
结束

函数 乘法(数一,数二)
{
	返回(数一 * 数二)
}

程序 显示(值)
	写格式("  显示() 收到参数 = %d\r\n", 值);
结束
