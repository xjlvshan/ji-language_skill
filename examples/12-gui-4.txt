程序 绘制图形
	整数 设备=取设备(窗体),画笔,红笔,画刷,红刷,字体1,行;
	画笔=创建画笔(0,1,$606060);
	红笔=创建画笔(0,3,$0000C0);
	画刷=创建画刷($F0F0F0);
	红刷=创建画刷($C0E0FF);
	字体1=新建字体(字号);
	选择对象(设备,画笔)选择对象(设备,画刷);
	画方块(设备,20,70,540,190);
	画方块(设备,20,290,540,355);
	选择对象(设备,红笔)选择对象(设备,红刷);
	画椭圆(设备,40,90,240,180);
	圆角矩形(设备,270,90,520,170,30,30);
	选择对象(设备,画笔)选择对象(设备,画刷);
	移到(设备,20,290);
	循环{画线(设备,20+行*5,290+行)行++}(行=104)
	选择对象(设备,红笔);
	移到(设备,20,345);
	循环{画线(设备,20+行*5,345-行)行++}(行=104)
	选择对象(设备,字体1);
	文本改色(设备,$800000);
	模式背景(设备,1);
	矩形类 文字框;
	文字框.左=30;文字框.顶=215;文字框.宽=520;文字框.高=40;
	绘制文本(设备,"GDI 绘图：画方块 / 画椭圆 / 画圆角矩形 / 画线 / 绘制文本",-1,文字框,0);
	选择对象(设备,新建字体(小字));
	文本改色(设备,$008000);
	文字框.左=30;文字框.顶=295;文字框.宽=520;文字框.高=30;
	绘制文本(设备,"字体：逻辑字体类 字号(高度=-26,名字=微软雅黑) + 新建字体",-1,文字框,0);
	文字框.左=30;文字框.顶=320;文字框.宽=520;文字框.高=30;
	绘制文本(设备,"消息：绘制按钮/绘制静态 + 发送消息(数据,指定字体,字体,0)",-1,文字框,0);
结束
