// ============================================================
// sys-3 文件与目录（逐条实测通过）
//  开文件/建文件/读文件/写文件/关文件/流定位/查找文件/下个文件/关闭查找
//  复制文件/文件改名/删除文件/创建目录/修改目录/删除目录/当前目录/模块名称/分区容量/分区空间
// 坑：①查找文件类要用 (长度=318) 初始化 ②.长度 才是文件大小(.大小 是高32位)
//     ③写格式("%s") 传结构体文本字段会栈溢出崩溃，文件名一律用 写文字
//     ④目录里 txt 文件很多(60+)时枚举会栈异常，示例用窄通配 sys3_*.txt
// ============================================================
文本 路径[256];
文本 内容[512];
文本 半[512];
查找文件类 项(长度=318);
整数 文件;
整数 长度;
整数 读了;
整数 写了;
整数 位置;
整数 下个;
整数 总数;
程序 初始启动
	写文字("=== 1. 取当前目录 / 模块名称 ===");
	写格式("当前目录 返回=%d 得到=",当前目录(256,路径));写文字(路径);
	写格式("模块名称 返回=%d 得到=",模块名称(0,路径,256));写文字(路径);
	写字符(10);
	写文字("=== 2. 建文件 / 写文件 / 关文件 ===");
	文件=建文件("sys3_test.txt",0);
	写格式("建文件 返回=%d",文件);写字符(10);
	如果(文件>0){
		复制文字(内容,"line1 hello");
		添字符(内容,13);
		添字符(内容,10);
		复制文字(半,"line2 world");
		添字符(半,13);
		添字符(半,10);
		加文字(内容,半);
		长度=文字长度(内容);
		写了=写文件(文件,内容,长度);
		写格式("写文件 请求=%d 实写=%d",长度,写了);写字符(10);
		关文件(文件);
	}否则{
		写文字("建文件失败");
	}
	写字符(10);
	写文字("=== 3. 开文件 / 流定位 / 读文件（方向:0开头 1当前 2结尾）===");
	文件=开文件("sys3_test.txt",0);
	写格式("开文件(方式0=只读) 返回=%d",文件);写字符(10);
	如果(文件>0){
		写格式("流定位(0,2) 文件长度=%d",流定位(文件,0,2));写字符(10);
		写格式("流定位(6,0) 返回=%d",流定位(文件,6,0));写字符(10);
		读了=读文件(文件,内容,5);
		内容(读了)=0;
		写格式("读文件 实读=%d 内容=",读了);写文字(内容);
		写格式("流定位(-6,2) 返回=%d",流定位(文件,-6,2));写字符(10);
		读了=读文件(文件,内容,6);
		内容(读了)=0;
		写格式("尾部读 实读=%d 内容=",读了);写文字(内容);
		关文件(文件);
	}
	写字符(10);
	写文字("=== 4. 查找文件 / 下个文件 / 关闭查找 ===");
	总数=0;下个=1;
	位置=查找文件("sys3_*.txt",项);
	写格式("查找文件 返回句柄=%d",位置);写字符(10);
	循环{
		写格式("  #%d 名称=",总数);写文字(项.名称);
		写格式("     长度=%d 属性=0x%X",项.长度,项.属性);写字符(10);
		总数++;
		下个=下个文件(位置,项);
	}(下个=0);
	写格式("sys3_*.txt 共 %d 个",总数);写字符(10);
	关闭查找(位置);
	写字符(10);
	写文字("=== 5. 复制文件 / 文件改名 / 删除文件 ===");
	删除文件("sys3_copy.txt");
	写格式("复制文件 返回=%d",复制文件("sys3_test.txt","sys3_copy.txt",假));写字符(10);
	写格式("文件改名 返回=%d",文件改名("sys3_copy.txt","sys3_ren.txt"));写字符(10);
	写格式("删除文件 返回=%d",删除文件("sys3_ren.txt"));写字符(10);
	写字符(10);
	写文字("=== 6. 创建目录 / 修改目录 / 删除目录 ===");
	删除目录("sys3_dir");
	写格式("创建目录 返回=%d",创建目录("sys3_dir",0));写字符(10);
	写格式("修改目录 返回=%d",修改目录("sys3_dir"));写字符(10);
	当前目录(256,路径);
	写文字("cwd=");
	写文字(路径);
	写格式("修改目录(..) 返回=%d",修改目录(".."));写字符(10);
	写格式("删除目录 返回=%d",删除目录("sys3_dir"));写字符(10);
	写字符(10);
	写文字("=== 7. 分区容量 / 分区空间（D盘）===");
	写格式("分区容量 返回=%d",分区容量("D:\\",@读了,@写了,@位置));写字符(10);
	写格式("  可用=%u 总=%u 剩余=%u",读了,写了,位置);写字符(10);
	写格式("分区空间 返回=%d",分区空间("D:\\",@读了,@写了,@位置,@总数));写字符(10);
	写格式("  每簇扇区=%d 扇区字节=%d 簇数=%d 可用簇=%d",读了,写了,位置,总数);写字符(10);
	写字符(10);
	写文字("=== 8. C运行库 remove（文件删除）===");
	写格式("文件删除(不存在的文件) 返回=%d",文件删除("no_such_file.tmp"));写字符(10);
	写文字("--- sys-3 演示结束 ---");
	退出(0);
结束
程序 添字符(文本 缓冲,整数 字符)
	整数 长=文字长度(缓冲);
	缓冲(长)=字符;
	缓冲(长+1)=0;
	返回(长+1)
结束
