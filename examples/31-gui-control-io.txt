// ============================================================
// 31-gui-control-io —— GUI 实战：控件内容读写 + 组合框 + 列表视图 + 剪贴板
// 全部使用实测可用的写法：
//   * 控件内容读写用官方运算符 ~   （文本变量 ~ 控件 读；控件 ~ 内容 写）
//   * 每个 程序 的局部声明必须写在所有语句之前
//   * 不在 如果(...) 条件里调用自定义过程（会 0xC0000005）
//   * 列表视图常量：插入列 $101B、插入项 $1007、写子项文字 $102E、取子项文字 $102D
//   * 组合框：组合添加 加项、$14E 置选中项、$147 取选中项
// 运行：编译后双击 exe，点「读取并加一行」按钮；实测能读到编辑框内容、
//       写入标签、往列表视图加行、并完成一次剪贴板写入+读回。
// ============================================================

结构 列结构{整数 掩码,格式,宽,文字,字长,子项,图,序;}
结构 项结构{整数 掩码,项,子项,状态,状态掩码,文字,字长,图,附加;}

整数 窗体,编辑,组合,列表,按钮,标签;
文本 缓冲[1024];
文本 提示[1024];
文本 剪切内容[1024];

程序 加载窗体
	窗体=创建窗口($100,程序.名称,"极语言-31-控件读写",$10C80000,100,100,720,520,0,0,0,0);
	编辑=创建窗口($200,"Edit","初始文字：极语言",$50211004,16,16,680,80,窗体,1,4,0);
	组合=创建窗口($200,"combobox","",$50210003,16,110,220,200,窗体,2,5,0);
	按钮=创建窗口($0,"Button","读取并加一行",$50000000,250,108,180,32,窗体,3,3,0);
	标签=创建窗口($0,"Static","（等待操作）",$50000000,16,150,680,24,窗体,4,2,0);
	列表=创建窗口($200,"SysListView32","",$50810001,16,184,680,300,窗体,5,0,0);
	发送消息(组合,组合添加,0,"棕色");
	发送消息(组合,组合添加,0,"深蓝");
	发送消息(组合,组合添加,0,"多彩");
	发送消息(组合,$14E,0,0);
	建表头;
	读取消息
结束

程序 窗体消息(窗口,消息,参数,数据)
	判断(消息)
	为 窗口关闭{停止(0)}
	为 单击左键{命令(参数)}
	否则{处理窗口(窗口,消息,参数,数据)}
结束

程序 命令(参数)
	整数 编号,序;
	编号=参数 & 65535;
	如果(编号=3)
	{
		缓冲 ~ 编辑;
		控件_写标签(缓冲);
		控件_加行(缓冲);
		序=发送消息(组合,$147,0,0);
		格式化(提示,"已读取：%s   组合框选中序号=%d",缓冲,序);
		控件_写标签(提示);
		测试剪贴板(缓冲);
	}
结束

程序 建表头
	放置(提示,"内容");列(0,200,提示);
	放置(提示,"长度");列(1,80,提示);
	放置(提示,"来源");列(2,120,提示);
结束

程序 列(序号,宽度,标题)
	列结构 列;
	列.掩码=3;
	列.格式=0;
	列.宽=宽度;
	列.文字=标题;
	列.字长=文字长度(标题);
	列.子项=序号;
	列.图=0;
	列.序=序号;
	发送消息(列表,$101B,序号,@列);
结束

整数 行数=0;

程序 控件_加行(内容)
	项结构 项;
	行数=行数+1;
	项.掩码=1;
	项.项=行数-1;
	项.子项=0;
	项.文字=内容;
	项.字长=文字长度(内容);
	发送消息(列表,$1007,0,@项);
	格式化(提示,"%d",文字长度(内容));
	项.子项=1;项.文字=提示;项.字长=文字长度(提示);
	发送消息(列表,$102E,行数-1,@项);
	放置(提示,"按钮");项.子项=2;项.文字=提示;项.字长=4;
	发送消息(列表,$102E,行数-1,@项);
结束

程序 控件_写标签(内容)
	标签 ~ 内容;
结束

程序 测试剪贴板(内容)
	整数 长,内存,址,句柄;
	长=文字长度(内容)+1;
	内存=全局分配($42,长);
	址=全局锁定(内存);
	内存复制(址,内容,长);
	全局解锁(内存);
	如果(开剪切板(0))
	{
		清剪切板;
		写剪切板(1,内存);
		关剪切板;
	}
	剪切内容(0)=0;
	如果(开剪切板(0))
	{
		句柄=读剪切板(1);
		如果(句柄<>0)
		{
			址=全局锁定(句柄);
			放置(剪切内容,址);
			全局解锁(句柄);
		}
		关剪切板;
	}
	句柄=文字长度(剪切内容);
	格式化(提示,"剪贴板读回 %d 字节",句柄);
	控件_写标签(提示);
结束

程序 放置(目标,字)
	整数 n;
	n=文字长度(字);
	内存复制(目标,字,n);
	目标(n)=0;
结束
