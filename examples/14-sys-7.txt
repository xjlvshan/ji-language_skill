// sys-7 哈希加密（MD5/SHA1）：按 inc\哈希加密.sc 的思路内联展开
// 依赖 advapi32.lib：加密服务/创建哈希/哈希数据/加密数据/销毁哈希/释放加密
程序 初始启动
	文本 数据[64];
	整数 服务;
	整数 哈希;
	整数 长度;
	复制文字(数据,"abc");
	服务=0;
	哈希=0;
	长度=0;
	加密服务(@服务,0,0,1,$F0000000);
	创建哈希(服务,$8003,0,0,@哈希);
	哈希数据(哈希,数据,3,0);
	加密数据(哈希,2,0,@长度,0);
	加密数据(哈希,2,数据,@长度,0);
	写格式("=== MD5(abc) 摘要长度=%d (期望16) ===",长度);写字符(10);
	写格式("字节0-7 = %d %d %d %d %d %d %d %d",数据(0),数据(1),数据(2),数据(3),数据(4),数据(5),数据(6),数据(7));写字符(10);
	写文字("标准值 = 0x90 0x01 0x50 0x98 0x3C 0xD2 0x4F 0xB0");写字符(10);
	销毁哈希(哈希);
	释放加密(服务,0);
	写字符(10);
	复制文字(数据,"abc");
	服务=0;
	哈希=0;
	长度=0;
	加密服务(@服务,0,0,1,$F0000000);
	创建哈希(服务,$8004,0,0,@哈希);
	哈希数据(哈希,数据,3,0);
	加密数据(哈希,2,0,@长度,0);
	加密数据(哈希,2,数据,@长度,0);
	写格式("=== SHA1(abc) 摘要长度=%d (期望20) ===",长度);写字符(10);
	写格式("字节0-7 = %d %d %d %d %d %d %d %d",数据(0),数据(1),数据(2),数据(3),数据(4),数据(5),数据(6),数据(7));写字符(10);
	写文字("标准值 = 0xA9 0x99 0x3E 0x36 0x47 0x06 0x81 0x6A");写字符(10);
	销毁哈希(哈希);
	释放加密(服务,0);
	写字符(10);
	写格式("算法常量 MD2=$%X MD4=$%X MD5=$%X SHA1=$%X SHA256=$%X SHA512=$%X",$8001,$8002,$8003,$8004,$800C,$800E);写字符(10);
	写文字("--- sys-7 演示结束 ---");
	退出(0);
