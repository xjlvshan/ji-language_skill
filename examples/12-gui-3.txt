模块文件=GUI.inc
整数 窗体,开始钮1,停钮2,进度3,标签4,菜单,菜单.1,菜单.2,计数;
程序 程序入口
列举 界面{.消息=@窗体消息
.名称="SecClass"
.光标=载入光标(0,32512)}
注册窗类(界面)窗口边界(桌面窗口,桌面)
加载窗体
结束
程序 读取消息
循环(获取消息(消息,0,0,0)){翻译消息(消息)分派消息(消息)}
结束
程序 加载窗体
	整数 左=(桌面.宽-560)>>1,上=(桌面.高-360)>>1;
	菜单=创建菜单
	菜单.1=创建菜单
	附加菜单(菜单,16,菜单.1,"文件")
	附加菜单(菜单.1,0,1001,"进度+20")
	附加菜单(菜单.1,$800,0,"")
	附加菜单(菜单.1,0,1002,"退出")
	菜单.2=创建菜单
	附加菜单(菜单,16,菜单.2,"帮助")
	附加菜单(菜单.2,0,1010,"关于")
	窗体=创建窗口($100,程序.名称,"极语言-gui-3-菜单与进度条",$10C80000,左,上,560,360,0,菜单,0,0);
	开始钮1=创建窗口($0,"Button","启动定时器(500ms)",$50000000,20,20,180,34,窗体,1,3,0);
	停钮2=创建窗口($0,"Button","停止定时器",$50000000,210,20,150,34,窗体,2,3,0);
	进度3=创建窗口($0,"msctls_progress32","",$50000000,20,80,500,28,窗体,3,10,0);
	标签4=创建窗口($0,"Static","定时事件(WM_TIMER=$113) 尚未启动；菜单编号从1001开始",$50000000,20,130,500,26,窗体,4,2,0);
	发送消息(进度3,进度限定,0,200);
	发送消息(进度3,进度位置,0,0);
	读取消息
结束
程序 窗体消息(窗口,消息,参数,数据)
	判断(消息)
	为 窗口关闭{停止(0)}
	为 单击左键{左键点击(参数)}
	为 定时事件{定时器消息(参数)}
	否则{处理窗口(窗口,消息,参数,数据)}
结束
程序 左键点击(参数)
	整数 编号=参数 & 65535,通知=参数 >> 16;
	文本 内容[128];
	判断(编号)
	为 1{设置定时(窗体,1,500,@定时器消息);改控件字(窗体,4,"已启动：设置定时(窗体,1,500,@定时器消息)")}
	为 2{设置定时(窗体,1,0);改控件字(窗体,4,"已停止：设置定时(窗体,1,0) 删除编号1的定时器")}
	为 1001{加进度}
	为 1002{停止(0)}
	为 1010{信息框(窗体,"定时器：设置定时(窗口,编号,毫秒,@回调程序)，毫秒=0 删除；菜单项目编号从 1001 起","关于",信息图标)}
	否则{格式化(内容,"编号=%d 通知码=%d 参数=%d",编号,通知,参数);改控件字(窗体,4,内容)}
结束
程序 加进度
	整数 位置=发送消息(进度3,进度取值,0,0);
	位置=位置+20;
	如果(位置>200)位置=0;
	发送消息(进度3,进度位置,位置,0);
	文本 内容[128];
	格式化(内容,"进度取值=%d 已加20",位置);
	改控件字(窗体,4,内容)
结束
程序 定时器消息(编号)
	计数=计数+1;
	整数 位置=发送消息(进度3,进度取值,0,0);
	位置=位置+5;
	如果(位置>200)位置=0;
	发送消息(进度3,进度位置,位置,0);
	文本 内容[128];
	格式化(内容,"定时器消息 编号=%d 触发次数=%d 进度=%d",编号,计数,位置);
	改控件字(窗体,4,内容)
结束
