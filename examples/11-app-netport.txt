程序类型=2
模块文件=CMD.inc,dll.inc
整数 端口表(6);
程序 初始启动
端口表(0)=3080;
端口表(1)=3306;
端口表(2)=5432;
端口表(3)=8080;
端口表(4)=135;
端口表(5)=445;
文本 数据[400];
整数 启=网络启动($202,数据);
写格式("Winsock=%d\r\n",启);
写文字("== 127.0.0.1 端口探测 ==");
文本 地址[16];
整数 i,网号,连,开放数=0;
循环于(i=0;i<6;i++)
{
网号=分配(2,1,6);
地址(0)%=2;
地址(2)%=反转(端口表(i));
地址(4)&=转换地址("127.0.0.1");
连=连网(网号,地址,16);
写格式("端口 %d : %d\r\n",端口表(i),连);
如果(连=0){开放数++;}
关闭(网号);
}
写格式("开放端口数=%d\r\n",开放数);
网络终止;
结束
