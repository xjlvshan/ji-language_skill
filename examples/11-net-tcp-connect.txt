模块文件=CMD.inc,dll.inc
程序 初始启动
文本 数据[400];
整数 启=网络启动($202,数据);
写格式("网络启动=%d\r\n",启);
整数 网号=分配(2,1,6);
文本 地址[16];
地址(0)%=2;
地址(2)%=反转(3080);
地址(4)&=转换地址("127.0.0.1");
整数 连=连网(网号,地址,16);
写格式("连 127.0.0.1:3080 = %d\r\n",连);
整数 网号2=分配(2,1,6);
地址(2)%=反转(9);
整数 连2=连网(网号2,地址,16);
写格式("连 127.0.0.1:9 = %d\r\n",连2);
关闭(网号);
关闭(网号2);
