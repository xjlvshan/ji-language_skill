程序 读取消息
循环(获取消息(消息,0,0,0)){翻译消息(消息)分派消息(消息)}
结束
程序 加载窗体
	整数 左=(桌面.宽-520)>>1,上=(桌面.高-340)>>1;
	窗体=创建窗口($100,程序.名称,"极语言-gui-1-骨架",$10C80000,左,上,520,340,0,0,0,0);
	标签1=创建窗口($0,"Static","消息显示区：请点按钮或窗体客户区",$50000000,20,20,470,44,窗体,1,2,0);
	按钮2=创建窗口($0,"Button","测窗口消息",$50000000,20,80,150,32,窗体,2,3,0);
	编辑3=创建窗口($200,"Edit","编辑框：单击/键盘消息会显示到上方标签",$50010000,20,130,470,26,窗体,3,4,0);
	改控件字(窗体,1,"启动完成：注册窗类/窗口边界/加载窗体/读取消息 全部执行");
	读取消息
结束
程序 窗体消息(窗口,消息,参数,数据)
	判断(消息)
	为 窗口关闭{停止(0)}
	为 单击左键{左键点击(参数,数据)}
	为 键盘按下{键盘事件(参数)}
	为 释放左键{左键点击(参数 & 65535,数据)}
	否则{处理窗口(窗口,消息,参数,数据)}
结束
程序 左键点击(参数,数据)
	整数 编号=参数 & 65535,通知=参数 >> 16;
	文本 内容[160];
	如果(编号=0)
		格式化(内容,"窗体客户区单击 参数=%d 数据=%d",参数,数据)
	否则
		格式化(内容,"控件编号=%d 通知码=%d 参数=%d",编号,通知,参数)
	改控件字(窗体,1,内容)
结束
程序 键盘事件(键码)
	文本 内容[160];
	格式化(内容,"键盘按下 虚拟键码=%d (0x%X) 键值常量见 0x41=A 0x25=左方向键",键码,键码)
	改控件字(窗体,1,内容)
结束
