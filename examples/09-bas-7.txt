程序 初始启动
	写文字("== bas-7 文本与数组 ==");

	写文字(内容);
	字节 码=内容(2);
	写格式("内容(2) = %d （'2' 的 ASCII）\r\n",码);
	内容(2)=\A;
	写文字(内容);
	内容(2)%=\好;
	写文字(内容);
	写格式("声明 文本 内容[64] 的 .长度 = %d\r\n",内容.长度);
	写文字("-- 文本是缓冲区：声明长度只是容量，写入超出会自动扩容 --");
	缓冲=缓冲+"12345678901234567890";
	写格式("声明 文本 缓冲[8] 的 .长度=%d，实际 文字长度() = %d\r\n",缓冲.长度,文字长度(缓冲));
	写文字(缓冲);
	写文字("-- 数组：固定长度 + 逐个赋值最稳 --");
	数列(0)=5;数列(1)=3;数列(2)=9;数列(3)=1;
	数列(4)=7;数列(5)=2;数列(6)=8;数列(7)=4;
	写格式("数列 长度=%d 数列(0)=%d 数列(7)=%d\r\n",数列.长度,数列(0),数列(7));
	整数 位=检索(数列,9,8);
	写格式("检索(数列,9,8) = %d\r\n",位);
	位=检索(数列,99,8);
	写格式("检索(数列,99,8) = %d （未找到返回 -1）\r\n",位);
	清零(数列,8,0);
	写格式("清零(数列,8,0) 后 数列(0)=%d\r\n",数列(0));
	列(1)=99;
	写格式("自动长度数组 列={10,20,30} 长度=%d 列(1)=%d\r\n",列.长度,列(1));
	写文字("-- 文本与数字互转 --");
	整数 数=转整数("1235");
	双精 双=转小数("12.34");
	写格式("转整数(1235) = %d\r\n",数);
	大缓冲(0)=0;
	大缓冲=大缓冲+"转小数(12.34) = "+双;
	写文字(大缓冲);
	格式化(大缓冲,"格式化 %.3f",双);
	写文字(大缓冲);
	写文字("程序结束");
