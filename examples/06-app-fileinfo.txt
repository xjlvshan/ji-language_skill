程序类型=2
模块文件=CMD.inc
程序 初始启动
// 1) 先写一个测试文件（建文件/写文件/关文件）
文本 样本[128];
内存复制(样本,"line one\r\nline two\r\nline three\r\n",31);
整数 出句=建文件("demo-out.txt",0);
如果(出句>0){
整数 写入=写文件(出句,样本,31);
关文件(出句);
写格式("写入字节=%d\r\n",写入);
}
// 2) 再读回来做统计（开文件/文件大小/读文件/申请内存）
整数 h=开文件("demo-out.txt",0);
整数 高位=0;
整数 长度=文件大小(h,@高位);
整数 内存=申请内存(长度+1);
整数 已读=读文件(h,内存,长度);
关文件(h);
内存(长度)=0;
写格式("文件=demo-out.txt 字节数=%d 已读=%d\r\n",长度,已读);
整数 i=0,行数=1,非空=0;
循环在(i<长度)
{
如果(内存(i)=10){行数++;}
如果(内存(i)>32){非空++;}
i++;
}
写格式("行数=%d 非空白字符=%d\r\n",行数,非空);
写文字("--- 内容 ---");
写字符(13);写字符(10);
写文字(内存);
释放内存(内存);
结束
