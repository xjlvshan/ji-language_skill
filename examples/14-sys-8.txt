// sys-8 实用工具：(a) 统计文本文件行数/字符数  (b) 列出指定目录下的 *.txt
// 说明：为避开"目录里 txt 太多导致枚举时栈缓冲区异常"的坑，本示例列的是 sys8_*.txt
文本 文件路径[256];
文本 缓存[8192];
文本 名称[64];
查找文件类 项(长度=318);
整数 文件;
整数 长度;
整数 读了;
整数 i;
整数 行数;
整数 字符数;
整数 搜;
整数 下个;
整数 数量;
程序 初始启动
	写文字("=== (a) 统计文本文件 ===");
	复制文字(文件路径,"sys8_demo.txt");
	文件=开文件(文件路径,0);
	写格式("开文件(sys8_demo.txt) 返回=%d",文件);写字符(10);
	如果(文件>0){
		长度=流定位(文件,0,2);
		流定位(文件,0,0);
		如果(长度>8000){长度=8000}
		读了=读文件(文件,缓存,长度);
		关文件(文件);
		写格式("文件长度=%d 实读=%d",长度,读了);写字符(10);
		行数=0;
		字符数=0;
		循环于(i=0;i<读了;i++){
			字符数++;
			如果(缓存(i)=10){行数++}
		}
		如果(读了>0){
			如果(缓存(读了-1)<>10){行数++}
		}
		写格式("字符数(字节)=%d",字符数);写字符(10);
		写格式("行数=%d",行数);写字符(10);
	}
	写字符(10);
	写文字("=== (b) 列出当前目录下 sys8_*.txt ===");
	数量=0;
	下个=1;
	搜=查找文件("sys8_*.txt",项);
	写格式("查找文件 返回=%d",搜);写字符(10);
	循环{
		写文字("  ");
		写文字(项.名称);
		写格式("   长度=%d 字节",项.长度);写字符(10);
		数量++;
		如果(数量>20){跳出}
		下个=下个文件(搜,项);
	}(下个=0);
	关闭查找(搜);
	写格式(" 共 %d 个文件",数量);写字符(10);
	写字符(10);
	写文字("=== (c) 生成一个统计结果文件 ===");
	文件=建文件("sys8_out.txt",0);
	如果(文件>0){
		复制文字(缓存,"统计结果: 行数/字符数见控制台输出");
		复制文字(名称,"sys8_demo.txt 已统计");
		长度=写文件(文件,缓存,文字长度(缓存));
		长度=写文件(文件,名称,文字长度(名称));
		关文件(文件);
		写文字("已写出 sys8_out.txt");
		写字符(10);
	}
	写文字("--- sys-8 演示结束 ---");
	退出(0);
结束
