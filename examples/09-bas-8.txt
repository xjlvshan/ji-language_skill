程序类型=2
模块文件=CMD.inc

常量 甲=1,乙=2,丙=3;
常量 红=0x10,蓝=0x20;
常量 样式=[甲+乙*2];
常量 位组合=[红|蓝];

文本 缓冲[128];

程序 初始启动
	写文字("== bas-8 小数/浮点/双精 转文本 与 写格式规则 ==");
	小数 单=1.5;
	浮点 浮=2.25;
	双精 双=3.125;
	写文字("[1] 直接 %%f 打印（关键结论）");
	写格式("  小数变量 单 %%f -> %f   <-- 错，打 0.000000\r\n",单);
	写格式("  浮点变量 浮 %%f -> %f   <-- 错，打 0.000000\r\n",浮);
	写格式("  双精变量 双 %%f -> %f   <-- 对\r\n",双);
	写格式("  双精字面量 3.5# %%f -> %f   <-- 对\r\n",3.5#);
	写文字("[2] 推荐写法A：拼进文本再 写文字()");
	缓冲(0)=0;
	缓冲=缓冲+"  小数="+单;
	写文字(缓冲);
	缓冲(0)=0;
	缓冲=缓冲+"  浮点="+浮;
	写文字(缓冲);
	缓冲(0)=0;
	缓冲=缓冲+"  双精="+双;
	写文字(缓冲);
	写文字("[3] 推荐写法B：格式化() 写进文本再 写文字()");
	格式化(缓冲,"  格式化 %.4f",双);
	写文字(缓冲);
	双精 提升=单;
	写格式("[4] 先把 小数 赋给 双精 再 %%f -> %f   <-- 对\r\n",提升);
	写文字("[5] 写格式 的格式串规则");
	写格式("  %%d=%d\r\n",255);
	写格式("  %%04X=%04X  %%o=%o  %%p=%p\r\n",255,8,4096);
	写格式("  [%%-5d]=[%-5d] [%%05d]=[%05d] [%%5d]=[%5d]\r\n",7,7,7);
	写格式("  %%c=%c  %%e=%e\r\n",65,12345.5#);
	写格式("  %%s 只接字面量: %s\r\n","hello");
	写文字("[6] 常量与编译期中括号");
	写格式("  常量 甲=%d 乙=%d 丙=%d\r\n",甲,乙,丙);
	写格式("  [甲+乙*2]=%d （中括号内按顺序算，不是先乘除）\r\n",样式);
	写格式("  [红|蓝]=%d\r\n",位组合);
	写格式("  [6+[35-12]*150-[60/12]]=%d\r\n",[6+[35-12]*150-[60/12]]);
	写文字("程序结束");
结束
