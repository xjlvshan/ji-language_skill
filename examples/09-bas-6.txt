程序 初始启动
	写文字("== bas-6 循环与跳转 =");
	整数 i=0;
	写文字("1. 循环(3) 固定次数");
	循环(3){写格式("  i=%d",i);i++;}
	写字符(10);
	写文字("2. 循环(条件) 条件成立就继续");
	i=0;
	循环(i<3){写格式("  i=%d",i);i++;}
	写字符(10);
	写文字("3. 循环{...} 无限循环 + 跳出");
	i=0;
	循环{i++;如果(i>4){跳出}写格式("  i=%d",i);}
	写字符(10);
	写文字("4. 循环于(i=0;i<3;i++) C 风格");
	循环于(i=0;i<3;i++){写格式("  i=%d",i);}
	写字符(10);
	写文字("5. 循环在(条件) while");
	i=0;
	循环在(i<3){写格式("  i=%d",i);i++;}
	写字符(10);
	写文字("6. 标记 起点 + 跳到 起点（无条件跳转）");
	i=0;
	标记 起点;
	i++;
	写格式("  i=%d",i);
	如果(i<3){跳到 起点;}
	写字符(10);
	跳到 终点;
	写文字("  这一行被 跳到 跳过，不会出现");
	标记 终点;
	写文字("  跳到 之后到达终点");
	写文字("7. 循环中 跳出");
	循环于(i=0;i<10;i++){如果(i=3){跳出}写格式("  i=%d",i);}
	写字符(10);
	写文字("程序结束");
