// ---------------- 输入即译：600ms 防抖 ----------------
// ★★ 绝对不要把“过程地址”传给 设置定时！
//    实测（src\_lab\tm5）`设置定时(窗体,号,毫秒,@过程)` 会在若干次回调后
//    以 0xC0000409（STATUS_STACK_BUFFER_OVERRUN）崩掉——编译器生成的
//    TIMERPROC 蹦床与 USER32 的调用约定/清栈方式不匹配，栈指针被逐步搞坏。
//    正确写法：第 4 个参数写 0（不注册回调），定时器就走标准 WM_TIMER，
//    由 窗体消息 的 `为 定时事件` 分支处理（src\_lab\tm4 实测 12s 计数 29，稳定）。
程序 排定翻译
	删除定时(窗体,号_防抖);
	设置定时(窗体,号_防抖,600,0);
结束

	立即翻译;
结束

// 载荷形如：<原文>\n\x1e\n<译文>
程序 拆分OCR(源,总长,甲,甲大小,乙,乙大小)
	整数 i,c,分隔位,起点,甲长;
	i=0;
	分隔位=-1;
	循环(i<总长)
	{
		c=源(i);
		如果(c=30){分隔位=i;i=总长}
		否则{i++}
	}
	如果(分隔位<0)
	{
		拷贝段(源,0,总长,甲,甲大小);
		乙(0)=0;
		返回(0)
	}
	甲长=分隔位;
	如果(甲长>0){如果(源(甲长-1)=10){甲长--}}
	拷贝段(源,0,甲长,甲,甲大小);
	起点=分隔位+1;
	如果(源(起点)=10){起点++}
	拷贝段(源,起点,总长,乙,乙大小);
	返回(1)
结束

程序 文字转整数(缓冲)
	整数 i,值,码,是数字;
	i=0;
	值=0;
	循环(缓冲(i)<>0)
	{
		码=缓冲(i);
		是数字=0;
		如果(码>47){是数字=1}
		如果(码>57){是数字=0}
		如果(是数字=1){值=值*10+(码-48)}
		i++;
	}
	返回(值)
