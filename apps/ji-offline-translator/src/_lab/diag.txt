程序 启动引擎

		改控件字(窗体,编号_状态,"未找到 Python 引擎");

	}
	放置(进程命令,路径甲);

	启动项.大小=68;
	拼路径(端口路径,根目录,"bridge_port.txt");
	文件删除(端口路径);
	整数 成功=创建进程(0,进程命令,0,0,0,$08000000,0,根目录,启动项,过程信息);
	引擎句柄=过程信息.进程;
	如果(引擎句柄=0){改控件字(窗体,编号_状态,"启动引擎失败");返回(-1)}

	整数 次数=0;
	循环(次数<600)
	{
		延时(200);
		次数++;
		整数 长度=读文字文件(端口路径,端口缓冲,64);
		如果(长度>0)
		{
			端口=文字转整数(端口缓冲);
			如果(端口>0){次数=9999}
		}
	}
	如果(端口=0){改控件字(窗体,编号_状态,"引擎未就绪");引擎句柄=0;返回(-1)}
	放置(提示,"引擎就绪，端口=");

程序 读文字文件(路径,缓冲,缓冲长)
	整数 句柄=文件打开(路径,"rb");

	如果(长度<0){长度=0}
	文件读取(缓冲,1,长度,句柄);
	文件关闭(句柄);
	缓冲(长度)=0;
	返回(长度)
