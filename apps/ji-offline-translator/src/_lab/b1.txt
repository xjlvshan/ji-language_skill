// ============ 文本工具 ============
程序 放置文字(目标,源)
	整数 n=文字长度(源);
	内存复制(目标,源,n);
	目标(n)=0;
结束

程序 文字转整数(缓冲)
	整数 i=0,值=0,码=0;
	循环(缓冲(i)<>0)
	{
		码=缓冲(i);
		如果(码>47)
		{
			如果(码<58)
			{
				值=值*10+(码-48);
			}
		}
		i++;
	}
	返回(值)
结束

程序 读文字文件(路径,缓冲,缓冲长)
	整数 句柄=文件打开(路径,"rb");
	如果(句柄=0){返回(-1)}
	定位文件(句柄,0,2);
	整数 长度=文件位置(句柄);
	定位文件(句柄,0,0);
	如果(长度>缓冲长-1){长度=缓冲长-1}
	如果(长度<0){长度=0}
	文件读取(缓冲,1,长度,句柄);
	文件关闭(句柄);
	缓冲(长度)=0;
	返回(长度)
结束

// ============ 编码转换 ============
程序 UTF8转本地(源,目标,目标大小)
	整数 宽数=到宽字符(65001,0,源,-1,0,0);
	如果(宽数<=0){目标(0)=0;返回(-1)}
	到宽字符(65001,0,源,-1,宽缓存,宽数);
	整数 n=到多字符(936,0,宽缓存,-1,目标,目标大小,0,0);
	返回(n)
结束

程序 本地转UTF8(源,源长,目标,目标大小)
	整数 宽数=到宽字符(936,0,源,源长,0,0);
	如果(宽数<=0){目标(0)=0;转换长度=0;返回(-1)}
	到宽字符(936,0,源,源长,宽缓存,宽数);
	整数 n=到多字符(65001,0,宽缓存,宽数,目标,目标大小,0,0);
	转换长度=n;
	返回(n)
结束

// ============ HTTP ============
程序 HTTP发送(端口,体,体长,响应,响应大小)
	文本 数据[400];
	文本 地址[16];
	文本 头[512];
	整数 启=网络启动($202,数据);
	整数 网=分配(2,1,6);
	如果(网<0){写文字("[错误] 分配套接字失败\r\n");返回(-1)}
	地址(0)%=2;
	地址(2)%=反转(端口);
	地址(4)&=转换地址("127.0.0.1");
	整数 连=连网(网,地址,16);
	如果(连<>0){关闭(网);网络终止;写文字("[错误] 连接失败\r\n");返回(-2)}
	格式化(头,"POST / HTTP/1.1\r\nHost: 127.0.0.1\r\nContent-Type: application/json\r\nContent-Length: %d\r\nConnection: close\r\n\r\n",体长);
	整数 头长=文字长度(头);
	发送(网,头,头长,0);
	如果(体长>0){发送(网,体,体长,0)}
	整数 总=0;
	循环(总<响应大小-1)
	{
		整数 n=接收(网,响应+总,响应大小-1-总,0);
		如果(n>0){总=总+n}
		否则{跳出}
	}
	关闭(网);
	网络终止;
	响应(总)=0;
	响应长度=总;
	返回(解析状态(响应,总))
结束

程序 解析状态(响应,总)
	整数 i=0,值=0,码=0,找到=0;
	循环(i<总)
	{
		如果(响应(i)=32){找到=1;i=总}
		如果(找到=0){i++}
	}
	如果(找到=1)
	{
		i++;
		找到=0;
		循环(i<总)
		{
			码=响应(i);
			如果(码<48){跳出}
			否则
			{
				如果(码>57){跳出}
				否则
				{
					值=值*10+(码-48);
					i++;
				}
			}
		}
	}
	返回(值)
结束
