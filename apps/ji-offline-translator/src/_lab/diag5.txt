	整数 n,i,有;

	拼路径(路径乙,根目录,"bridge.py");
	有=有文件(路径甲);
	如果(有=1){返回(0)}
	拼路径(路径甲,根目录,"..\engine\python.exe");
	拼路径(路径乙,根目录,"..\bridge.py");
结束

程序 启动引擎
	整数 有,成功,次数,长度;
	如果(端口>0){返回(0)}
	有=有文件(路径甲);
	如果(有=0)
	{
		改控件字(窗体,编号_状态,"未找到 Python 引擎");
		返回(-1)
	}
	格式化(进程命令,"%s %s",路径甲,路径乙);
	启动项.大小=68;
	放置(提示,路径乙);
	取目录(提示);
	拼路径(端口路径,提示,"bridge_port.txt");
	文件删除(端口路径);
	成功=创建进程(0,进程命令,0,0,0,$08000000,0,根目录,启动项,过程信息);
	引擎句柄=过程信息.进程;
	如果(引擎句柄=0){改控件字(窗体,编号_状态,"启动引擎失败");返回(-1)}
	改控件字(窗体,编号_状态,"离线翻译引擎已启动，正在等待就绪...");
	返回(0)
结束

程序 确保端口
	整数 有,次数;
	如果(端口>0){返回(1)}
	次数=0;
	循环(次数<40)
	{
		有=有文件(端口路径);
		如果(有=1){跳出}
		延时(150);
		次数++;
	}
	读文字文件(端口路径,端口缓冲,64);
	端口=文字转整数(端口缓冲);
	如果(端口>0){返回(1)}
	改控件字(窗体,编号_状态,"翻译引擎未就绪");
	返回(0)
结束

程序 读文字文件(路径,缓冲,缓冲长)
	整数 句柄,长度;
	句柄=文件打开(路径,"rb");
	如果(句柄=0){缓冲(0)=0;返回(-1)}
	定位文件(句柄,0,2);
	长度=文件位置(句柄);
	定位文件(句柄,0,0);
	如果(长度>缓冲长-1){长度=缓冲长-1}
	如果(长度<0){长度=0}
	文件读取(缓冲,1,长度,句柄);
	文件关闭(句柄);
	缓冲(长度)=0;
	返回(长度)
结束

程序 报转义(内容)
	整数 i,c,转义,原样;
	i=0;
	循环
	{
		c=内容(i);
		如果(c=0){跳出}
		转义=0;
		原样=1;
		如果(c=34){转义=1}
		如果(c=92){转义=1}
		如果(c=10){转义=1}
		如果(c=13){转义=1}
		如果(c=10){原样=0}
		如果(c=13){原样=0}
		如果(转义=1){报字节(92)}
		如果(c=10){报字节(110)}
		如果(c=13){报字节(114)}
		如果(原样=1){报字节(c)}
		i++;
	}
结束

// ---------------- 编码转换（GBK <-> UTF-8） ----------------
程序 本地转UTF8(源,源长,目标,目标大小)
	整数 宽数,n;
	宽数=到宽字符(936,0,源,源长,0,0);
	如果(宽数<=0){目标(0)=0;转换长度=0;返回(-1)}
	到宽字符(936,0,源,源长,宽缓存,宽数);
	n=到多字符(65001,0,宽缓存,宽数,目标,目标大小,0,0);
	转换长度=n;
	返回(n)
结束

程序 UTF8转本地(源,目标,目标大小)
	整数 宽数,n;
	宽数=到宽字符(65001,0,源,-1,0,0);
	如果(宽数<=0){目标(0)=0;转换结果长=0;返回(-1)}
	到宽字符(65001,0,源,-1,宽缓存,宽数);
	n=到多字符(936,0,宽缓存,-1,目标,目标大小,0,0);
	转换结果长=n;
	返回(n)
结束
