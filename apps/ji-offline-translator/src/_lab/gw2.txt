程序 本地转UTF8(源,源长,目标,目标大小)
	整数 宽数,n;
	宽数=到宽字符(936,0,源,源长,0,0);
	如果(宽数<=0){目标(0)=0;转换长度=0;返回(-1)}
	到宽字符(936,0,源,源长,宽缓存,宽数);
	n=到多字符(65001,0,宽缓存,宽数,目标,目标大小,0,0);
	转换长度=n;
	返回(n)
结束

程序 UTF8转本地(源,目标,目标大小)
	整数 宽数,n;
	宽数=到宽字符(65001,0,源,-1,0,0);
	如果(宽数<=0){目标(0)=0;转换结果长=0;返回(-1)}
	到宽字符(65001,0,源,-1,宽缓存,宽数);
	n=到多字符(936,0,宽缓存,-1,目标,目标大小,0,0);
	转换结果长=n;
	返回(n)
结束

程序 文字转整数2(缓冲)
	整数 i,值,码,是数字;
	i=0;
	值=0;
	循环(缓冲(i)<>0)
	{
		码=缓冲(i);
		是数字=0;
		如果(码>47){是数字=1}
		如果(码>57){是数字=0}
		如果(是数字=1){值=值*10+(码-48)}
		i++;
	}
	返回(值)
结束

程序 HTTP发送(端口号,体,体长,响应,响应大小)
	文本 数据[400];
	文本 地址[16];
	文本 头[512];
	整数 网,连,总,n;
	网络初始化状态=网络启动($202,数据);
	网=分配(2,1,6);
	分配状态=网;
	如果(网<0){返回(-1)}
	地址(0)%=2;
	地址(2)%=反转(端口号);
	地址(4)&=转换地址("127.0.0.1");
	连=连网(网,地址,16);
	连网状态=连;
	如果(连<>0){关闭(网);网络终止;返回(-2)}
	格式化(头,"POST / HTTP/1.0\r\nHost: 127.0.0.1\r\nContent-Type: application/json\r\nContent-Length: %d\r\nConnection: close\r\n\r\n",体长);
	发送(网,头,文字长度(头),0);
	如果(体长>0){发送(网,体,体长,0)}
	总=0;
	循环(总<响应大小-1)
	{
		n=接收(网,响应+总,响应大小-1-总,0);
		如果(n>0){总=总+n}
		否则{跳出}
	}
	关闭(网);
	网络终止;
	响应(总)=0;
	响应长度=总;
	返回(总)
结束
