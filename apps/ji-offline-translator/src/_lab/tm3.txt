模块文件=GUI.inc
定义 后缀=exe;
// 最小定时器探针：验证 GUI 程序里 设置定时 + 回调 本身是否可用
常量 编号_标签=1;
常量 号_定时=1;
整数 窗体,控件_标签,计数=0;
文本 提示[128];

程序 加载窗体
	整数 左,上;
	左=200;
	上=200;
	窗体=创建窗口($100,"SecClass","定时器最小探针",$10C80000,左,上,420,220,0,0,0,0);
	控件_标签=创建窗口($0,"Static","计数=0",$50000000,20,20,360,40,窗体,编号_标签,2,0);
	设置定时(窗体,号_定时,500,@定时到);
	读取消息
结束

程序 窗体消息(窗口,消息,参数,数据)
	判断(消息)
	为 窗口关闭{停止(0)}
	为 定时事件{定时到(0)}
	否则{处理窗口(窗口,消息,参数,数据)}
结束

程序 定时到(编号)
	计数++;
	格式化(提示,"计数=%d",计数);
	改控件字(窗体,编号_标签,提示);
结束
