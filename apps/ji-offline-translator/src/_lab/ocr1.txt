模块文件=GUI.inc
定义 后缀=exe;
// OCR 抓屏探针：不交互，启动 800ms 后抓 (100,100)-(400,300) 存 BMP，然后退出。
// 用来单独验证：创建设备/创建图像/位图传输/设备位图(逐行)/文件写入/BMP 头
结构 图形信息类{整数 长度,宽度,高度;整形 层数,位数;整数 压缩,图大小,水平,垂直,实际,重要;}
整数 窗体,抓屏成功=0,行数=0;
文本 行缓冲[9604];
文本 全路径[512];
图形信息类 图形信息;

程序 加载窗体
	窗体=创建窗口($100,程序.名称,"OCR抓屏探针",$10C80000,100,100,520,200,0,0,0,0);
	延时(800);
	放置(全路径,"D:\SEC\project\ji-offline-translator\src\_lab\ocr_probe.bmp");
	抓屏(100,100,300,200);
	停止(0)
结束

程序 窗体消息(窗口,消息,参数,数据)
	处理窗口(窗口,消息,参数,数据)
结束

程序 抓屏(左,上,宽,高)
	整数 设备,图层,图像,行字节,总,行,句柄;
	抓屏成功=0;
	行字节=宽*3;
	行字节=((行字节+3)>>2)<<2;
	总=行字节*高;
	设备=取设备(0);
	图层=创建设备(设备);
	图像=创建图像(设备,宽,高);
	选择对象(图层,图像);
	位图传输(图层,0,0,宽,高,设备,左,上,$00CC0020);
	图形信息.长度=40;
	图形信息.宽度=宽;
	图形信息.高度=高;
	图形信息.层数=1;
	图形信息.位数=24;
	图形信息.压缩=0;
	图形信息.图大小=0;
	图形信息.水平=0;
	图形信息.垂直=0;
	图形信息.实际=0;
	图形信息.重要=0;
	句柄=文件打开(全路径,"wb");
	如果(句柄=0){返回(0)}
	写图头(句柄,宽,高,行字节,总);
	行=0;
	循环(行<高)
	{
		设备位图(图层,图像,行,1,行缓冲,@图形信息,0);
		文件写入(行缓冲,1,行字节,句柄);
		行++;
	}
	文件关闭(句柄);
	选择对象(图层,0);
	删除对象(图像);
	删除设备(图层);
	释放设备(0,设备);
	行数=行;
	抓屏成功=1;
结束

程序 写图头(句柄,宽,高,行字节,总)
	文本 头[64];
	整数 大小;
	大小=54+总;
	头(0)=66;
	头(1)=77;
	写四字节(头,2,大小);
	写四字节(头,6,0);
	写四字节(头,10,54);
	写四字节(头,14,40);
	写四字节(头,18,宽);
	写四字节(头,22,高);
	头(26)=1;
	头(27)=0;
	头(28)=24;
	头(29)=0;
	写四字节(头,30,0);
	写四字节(头,34,0);
	写四字节(头,38,0);
	写四字节(头,42,0);
	写四字节(头,46,0);
	写四字节(头,50,0);
	文件写入(头,1,54,句柄);
结束

程序 写四字节(缓冲,位,值)
	整数 甲,乙,丙,丁;
	甲=值 & 255;
	乙=值 >> 8;
	乙=乙 & 255;
	丙=值 >> 16;
	丙=丙 & 255;
	丁=值 >> 24;
	丁=丁 & 255;
	缓冲(位)=甲;
	缓冲(位+1)=乙;
	缓冲(位+2)=丙;
	缓冲(位+3)=丁;
结束

程序 放置(目标,字)
	整数 n;
	n=文字长度(字);
	内存复制(目标,字,n);
	目标(n)=0;
结束
